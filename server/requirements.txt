# DropSwift server dependencies.
#
# Required on Windows (and recommended on macOS/Linux) so the iPhone app can
# auto-discover this computer on the local network via Bonjour/mDNS.
# On macOS the server also falls back to the built-in `dns-sd` if this is absent.
#
#   pip install -r requirements.txt
#
zeroconf>=0.130

# Optional: prints a scannable QR code (same pairing link the Mac app shows as
# an image) to the console so console-only launches — Windows, or a manual
# `python3 server.py` — can also be paired by scanning instead of typing the
# access code. The server runs fine without it, just without the QR.
qrcode>=7.4
